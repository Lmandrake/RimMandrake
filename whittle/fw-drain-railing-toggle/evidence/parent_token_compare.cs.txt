using System;
using System.IO;
using System.Linq;
using Microsoft.CodeAnalysis.CSharp;

string Read(string path)
{
    string text = File.ReadAllText(path);
    if (path.EndsWith(".targets"))
    {
        text = text.Split("<![CDATA[")[1].Split("]]>")[0];
        text = "class Extractor { bool Execute() {" + text + "return true; } }";
    }
    return text;
}
