def kv(txt, key):
    m = re.search(r"\b%s=(\S+)" % re.escape(key), txt or "")
    return m.group(1) if m else None


def parse_tiers(txt):

        out[name] = {k: kv(part, k) for k in ("tier", "adultTier", "bodySize", "wakeComp")}
    return out

class Run(object):
    def __init__(self, rb, out):
        self.rb, self.out, self.rows, self.log = rb, out, [], []
        self.required_rows = ROWS

    def call(self, tool, **kw):
        try:
            r = self.rb.call(tool, kw) or {}
        except Exception as ex:  # noqa: BLE001
            r = {"success": False, "error": "call raised: %s" % ex}
        self.log.append((tool, kw, str(r)[:400]))
        return r

    def hook(self, method, args=""):
        r = self.call("jawa/static_call", type=HOOKS, method=method, args=args)
        txt = str(r.get("result") if r.get("result") is not None else r.get("error") or r.get("message") or r)
        print("    %s(%s) -> %s" % (method, args, txt[:400]), flush=True)
        return r.get("success") is True and not txt.startswith("FAIL"), txt

    def get(self, f):
        return self.call("jawa/mod_settings_field", typeName=SETTINGS, action="get", field=f).get("value")

    def put(self, f, v):
        # restore-check: ok -- main() snapshots FIELDS first and puts every one back in its finally
        return self.call("jawa/mod_settings_field", typeName=SETTINGS, action="set", field=f, value=str(v)).get("success")

    def ticks(self, n):
        done = 0
        while done < n:
            k = min(1000, n - done)
            self.call("rimworld/step_game_ticks", ticks=k)
            done += k

    def row(self, rid, status, cls, detail):
        self.rows.append(dict(id=rid, item=ITEM, criterion="A2", status=status, cls=cls, detail=detail))
        print("  %-42s %-10s %-7s %s" % (rid, status, cls, detail[:300]), flush=True)
        self.save()

    def save(self):
        os.makedirs(os.path.dirname(self.out), exist_ok=True)
        with open(self.out, "w", encoding="utf-8") as f:
            json.dump(dict(script="titan_test_races_live.py", item=ITEM, when=time.strftime("%Y-%m-%dT%H:%M:%S"), rows=self.rows,
                           calls=self.log, summary=aggregate(self.rows, self.required_rows)), f, indent=1, default=str)

def judge_yield(t2, ctl):
    """(status, detail) from two ButcherYieldReport lines. Each of n pawns rounds its meat once (GenMath.RoundRandom), so the
    T2 total sits within n of control x factor."""
    try:
        n = int(kv(t2, "n"))
        m2, mc, f = int(kv(t2, "meat")), int(kv(ctl, "meat")), float(kv(t2, "factor"))
        fc = float(kv(ctl, "factor"))
    except (TypeError, ValueError):
        return None, "unparsable: %s || %s" % (t2, ctl)
    d = "T2 meat=%d control meat=%d factor=%.4f (control factor %.4f) expected %.1f +- %d" % (m2, mc, f, fc, mc * f, n)
    if kv(ctl, "tier") != "None" or fc != 1.0:
        return "FAIL", "control is tiered or curved: " + d
    if mc <= 0:
        return None, "control butchered no meat: " + d
    if f == 1.0:
        # SubLinearFactor = sqrt(floor / bodySize): a subject ON its tier floor (the T2 test race is bodySize 8 = the T2 floor)
        # gets 1 by design, so equal meat proves nothing about the curve. Not a MOD FAIL: the row has no reducing subject
        # (gen 7, live 2026-11-03: T2 6720 == control 6720 at factor 1.0000 was filed as MOD FAIL and passed A2)
        return None, "no reducing subject: the kernel factor is 1 (a T1/T2 body above its tier floor is needed): " + d
    if abs(f - YIELD_FACTOR) > 1e-3:
        return "FAIL", "kernel factor is not sqrt(8/12) for a bodySize-12 T2: " + d
    ok = f < 1.0 and m2 < mc and abs(m2 - mc * f) <= n
    return ("PASS" if ok else "FAIL"), d


def judge_roof(t2, ctl, mode=None):
    """Judge measured Position-setter integration, never a walked path/clearing job.

    Mode is raw setting readbacks, not a caller's label. Unknown/missing/malformed
    measurements have no verdict. Wrong measured outcomes FAIL in either mode.
    """
    if not isinstance(mode, dict) or any(not isinstance(mode.get(f), str) or
            mode[f].lower() not in ("true", "false") for f in ROOF_FIELDS):
        return None, "unmeasured/malformed roof-wall mode: %r" % mode
    if any(mode[f].lower() != "true" for f in ROOF_FIELDS if f != "wallsSlotsEnabled"):
        return None, "roof/wake/clearing prerequisites disabled: %r" % mode
    active = all(mode[f].lower() == "true" for f in
                 ("giantAnimalsEnabled", "wakeEnabled", "titanClearingEnabled", "wallsSlotsEnabled"))

    def parse(txt):
        if not isinstance(txt, str):
            return None
        match = re.fullmatch(r"(\w+) tier=(\w+) roofAfter=(\w+) wallHpBefore=(\d+) "
                             r"wallHpAfter=(\d+) wallDestroyed=(True|False)", txt)
        if not match:
            return None
        name, tier, roof, before, after, destroyed = match.groups()
        before, after = int(before), int(after)
        gone = destroyed == "True"
        if before <= 0 or not 0 <= after <= before or gone != (after == 0):
            return None
        return name, tier, roof, before, after, gone

    t, c = parse(t2), parse(ctl)
    if t is None or c is None:
        return None, "invalid/non-positive/inconsistent probe data: %s || %s" % (t2, ctl)
    d = ("Position-setter teleport L1 (no path/job); WallsSlotsActive=%s; T2 roof=%s wall %d->%d "
         "destroyed=%s; forced-out control roof=%s wall %d->%d destroyed=%s" %
         (active, t[2], t[3], t[4], t[5], c[2], c[3], c[4], c[5]))
    identity = t[:2] == (T2, "T2") and c[:2] == (CTL, "None")
    wall_ok = (t[4] == t[3] and not t[5]) if active else t[4] < t[3]
    control_ok = c[2] == "RoofConstructed" and c[4] == c[3] and not c[5]
    return ("PASS" if identity and t[2] == "none" and wall_ok and control_ok else "FAIL"), d


def roofwall_row(R, cells):
    """Reusable bounded R4: both actual modes, a T2 + forced-out control per mode.

    Snapshot raw settings before any write; read all mode fields before and after
    each call. Each arm needs both successful probes and a stable measured mode.
    Restoring settings is part of this row's pass condition.
    """
    prior = {f: R.get(f) for f in ROOF_FIELDS}
    arms, errors = [], []
    status = "UNMEASURED"
    if any(not isinstance(v, str) or v.lower() not in ("true", "false") for v in prior.values()):
        R.row(ROOF_ROW, status, "HARNESS", "cannot snapshot raw roof settings: %r" % prior)
        return
    try:
        for slots in (True, False):
            expected = {f: str(slots if f == "wallsSlotsEnabled" else True) for f in ROOF_FIELDS}
            if any(not R.put(f, v) for f, v in expected.items()):
                raise RuntimeError("mode write refused")
            probes = []
            for kind in (T2, CTL):
                before = {f: R.get(f) for f in ROOF_FIELDS}
                ok, txt = False, None
                for x, z in cells:
                    ok, txt = R.hook("RoofWallProbe", "%s|%d,%d" % (kind, x, z))
                    if ok or "not usable" not in txt:
                        break
                after = {f: R.get(f) for f in ROOF_FIELDS}
                stable = before == after and all(isinstance(before.get(f), str) and
                         before[f].lower() == expected[f].lower() for f in ROOF_FIELDS)
                probes.append(dict(kind=kind, ok=ok, raw=txt, before=before, after=after, stable=stable))
            if all(p["ok"] and p["stable"] for p in probes):
                st, detail = judge_roof(probes[0]["raw"], probes[1]["raw"], probes[0]["before"])
            else:
                st, detail = None, "probe failed or actual settings missing/changed"
            arms.append(dict(mode="default Walls/Slots" if slots else "legacy wake", probes=probes,
                             status=st or "UNMEASURED", detail=detail))
        status = "FAIL" if any(a["status"] == "FAIL" for a in arms) else (
                 "PASS" if len(arms) == 2 and all(a["status"] == "PASS" for a in arms) else "UNMEASURED")
    except Exception as ex:
        errors.append(str(ex))
    finally:
        for f, v in prior.items():
            try:
                if not R.put(f, v) or R.get(f) != v:
                    errors.append("raw restore mismatch: " + f)
            except Exception as ex:
                errors.append("restore %s: %s" % (f, ex))
        if errors:
            status = "UNMEASURED"
        R.roofwall_evidence = dict(prior=prior, arms=arms, errors=errors, status=status)
        R.row(ROOF_ROW, status, "MOD" if status in ("PASS", "FAIL") else "HARNESS",
              "Position-setter L1 only; " + json.dumps(R.roofwall_evidence, sort_keys=True))


def judge_thick(t2, ctl):

def body(R, cells):
    ok, txt = R.hook("TierReport", "%s,%s,%s,%s,%s,Elephant" % (T2, T3, CTL, T2Y, CTLY))
    rep = parse_tiers(txt) if ok else {}
    if not ok or any(k not in rep or rep[k].get("tier") is None for k in (T2, T3, CTL, T2Y, CTLY)):
        R.row("t2_t3_tiers", "UNMEASURED", "HARNESS", "test races not readable (is mandrake.rm.hugethingstestraces on the list?): "
              + txt[:200])
        return
    bad = judge_tiers(rep)
    R.row("t2_t3_tiers", "FAIL" if bad else "PASS", "MOD", ("wrong: %s; " % bad if bad else "") + txt[:300])

    # R4 / R5 first: they need clean open cells, and R2 leaves corpses and sites behind.
    def probe(method, kind):

            if "not usable" not in t:

def corpse_site_row(R, cells):

NO_VERDICT = 2      # exit code: the proof never reached the game (not PASS 0, not FAIL 1); the live round reruns/restages


def open_bridge():
    """-> a connected RimBridge, or None after printing why. An unreachable bridge (WSL python3, no game, no log) is a
    harness fact that says nothing about the mod, so the caller exits NO_VERDICT, never FAIL: two A2 'FAIL's
    (2026-11-02T20:21Z, 2026-11-03T06:01Z) were this traceback exiting 1."""
    from rimbridge_client import RimBridge, RimBridgeError, discover_from_log
    try:
        host, port, token = discover_from_log()
        rb = RimBridge(host, port, token, timeout=180.0)
        rb.connect()
        return rb
    except (RimBridgeError, OSError) as e:
        print("UNMEASURED: the bridge could not be reached, no verdict (exit %d): %s" % (NO_VERDICT, str(e)[:400]), flush=True)
        return None


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--site", default="", help="first probe cell X,Z (default: map centre, then offsets)")
    ap.add_argument("--out", default=os.path.join(REPO, "Transient", "modcheck",
                                                  "titan_test_races_%s.json" % time.strftime("%Y%m%dT%H%M%S")))
    a = ap.parse_args()
    rb = open_bridge()
    if rb is None:
        return NO_VERDICT
    with rb:
        R = Run(rb, a.out)
        snap = dict((f, R.get(f)) for f in FIELDS)
        if any(v is None for v in snap.values()):
            R.row("settings_snapshot", "UNMEASURED", "HARNESS", "could not read %r" % snap)
            print("rows -> %s" % a.out)
            return 1
        mi = R.call("jawa/map_info")
        size = mi.get("size") or mi.get("mapSize") or {}
        try:
            cx, cz = int(size.get("x")) // 2, int(size.get("z")) // 2
        except (TypeError, ValueError, AttributeError):
            cx = cz = 125
        if a.site:
            cx, cz = [int(v) for v in a.site.split(",")]
        cells = [(cx + dx, cz + dz) for dx, dz in ((0, 0), (12, 0), (-12, 0), (0, 12), (0, -12), (12, 12), (-12, -12), (20, 20))]
        try:
            unset = [f for f, v in FIELDS.items() if not R.put(f, v) or str(R.get(f)).lower() != str(v).lower()]
            custom = R.get("tierThresholdsCustom")
            if str(custom).lower() != "false":
                # a startup-read setting: the expected tiers are the default 4/8/20 ladder, so a custom one cannot be judged here
                R.row("settings_ladder", "UNMEASURED", "SITE", "tierThresholdsCustom=%r: the ladder is not the default 4/8/20" % custom)
            elif unset:
                R.row("settings_set", "UNMEASURED", "HARNESS", "could not set %r: no row can be judged" % unset)
            else:
                body(R, cells)
        finally:
            lost = [f for f, v in snap.items() if not R.put(f, v) or str(R.get(f)).lower() != str(v).lower()]
            if lost:
                print("RESTORE FAILED for %r (wanted %r)" % (lost, dict((f, snap[f]) for f in lost)), flush=True)
            R.save()
    print("rows -> %s" % a.out)
    v = verdict(R.rows)
    print("ALL ROWS CLASSIFIED: %s" % (classification_verdict(R.rows) == 0))
    print("GAMEPLAY VERDICT: %s (%d rows)" % ("PASS" if v == 0 else "FAIL", len(R.rows)))
    return v

def defs_in(folder):
    out = []
    for dp, _, fs in os.walk(folder):

def resolve(nodes, tag):
    """defName -> (effective baseBodySize, forceEnabled or None, harnessOnly); ParentName chains inside `nodes` only."""
    by_name = {n.get("Name"): n for n in nodes if n.tag == tag and n.get("Name")}

    def field(n, path):
        while n is not None:
            v = n.findtext(path)
            if v is not None:
                return v.strip()
            n = by_name.get(n.get("ParentName"))
        return None

    out = {}
    for n in nodes:
        if n.tag != tag or not n.findtext("defName"):
            continue
        bs = field(n, "race/baseBodySize")
        fe, ho = None, False
        for li in n.findall("modExtensions/li"):
            if li.get("Class") == EXT and li.findtext("forceEnabled") is not None:
                fe = li.findtext("forceEnabled").strip().lower() == "true"
            if li.get("Class") == EXT and (li.findtext("harnessOnly") or "").strip().lower() == "true":
                ho = True
        out[n.findtext("defName").strip()] = (float(bs) if bs else None, fe, ho)
    return out


def tier_for(size, t1, t2, t3, force):
    # RM_TitanicKernel.TierFor: ForceOut -> None; >= boundaries; ForceIn floors at T1.
    if force is False:
        return "None"
    tier = "T3" if size >= t3 else "T2" if size >= t2 else "T1" if size >= t1 else "None"
    return "T1" if tier == "None" and force is True else tier

def roofwall_modes():

    import titan_test_races_live as L

        def __init__(self, fault=""):
            self.settings = {f:"False" for f in L.ROOF_FIELDS}
            self.initial = dict(self.settings)
            self.rows, self.probes, self.fault = [], [], fault
        def get(self, f):
            return self.settings[f]
        def put(self, f, v):
            self.settings[f] = str(v)
            return True
        def hook(self, method, args):
            self.probes.append((method,args,dict(self.settings)))
            control = args.startswith(L.CTL + "|")
            legacy = self.settings["wallsSlotsEnabled"] == "False"
            if self.fault == "refused control" and control:
                return False, "FAIL missing kind"
            hp = 175 if legacy and not control and self.fault != "always intact" else 195
            if self.fault == "mode drift":
                self.settings["wallsSlotsEnabled"] = "False"
            return True, ("%s tier=%s roofAfter=%s wallHpBefore=195 wallHpAfter=%d wallDestroyed=False" %
                (L.CTL if control else L.T2, "None" if control else "T2", "RoofConstructed" if control else "none", hp))

            self.rows.append(dict(id=rid,status=status,cls=cls,detail=detail))
